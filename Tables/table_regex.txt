// Normalize multiple spaces to single space
[ ]{2,} --> " "

// Phone numbers to format: XXX-XXXXXX
([0-9]{2})[ \-]?([0-9]{6}) --> $1-$2

// Dates: YYYY/MM/DD or YYYY-MM-DD to DD.MM.YYYY
([0-9]{4})[/-]([0-9]{2})[/-]([0-9]{2}) --> $3.$2.$1

// Orders: normalize "Cxxx" and "codexxx" to uppercase
(C|c|Code|code)\.?[ ]?([0-9]{1,3}) --> C$2
